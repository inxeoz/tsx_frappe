Available API Methods:

  • window.ITSupportTheme.setTheme('theme-name') - Set specific theme
  • window.ITSupportTheme.toggle() - Toggle between light/dark
  • window.ITSupportTheme.cycle() - Cycle through all themes
  • window.ITSupportTheme.getCurrentTheme() - Get current theme
  • ThemeControl.setTimeBasedTheme() - Apply time-based theme
  • ThemeControl.detectSystem() - Detect system preference

Integration Examples

JavaScript Integration:

// Set theme programmatically
if (window.ITSupportTheme) {
    window.ITSupportTheme.setTheme('blue');
}

// Schedule theme change
setTimeout(() => {
    window.ITSupportTheme.setTheme('dark');
}, 5000);

// User preference based theme
const userTheme = localStorage.getItem('user-theme') || 'system';
window.ITSupportTheme.setTheme(userTheme);

Frappe Integration:

// In Frappe DocType JavaScript
frappe.ready(function() {
    if (window.ITSupportTheme) {
        // Set theme based on user preference
        const theme = frappe.user.theme || 'system';
        window.ITSupportTheme.setTheme(theme);

        // Add theme selector to toolbar
        frappe.ui.toolbar.add_dropdown_button('Theme', 'palette', [
            {
                label: 'Light Theme',
                action: () => window.ITSupportTheme.setTheme('light')
            },
            {
                label: 'Dark Theme',
                action: () => window.ITSupportTheme.setTheme('dark')
            }
        ]);
    }
});

Department-Based Themes:

// Auto-apply themes based on user department
const departmentThemes = {
    'IT': 'blue',
    'HR': 'green',
    'Finance': 'purple',
    'Management': 'high-contrast'
};

const userDept = getCurrentUserDepartment();
const theme = departmentThemes[userDept] || 'system';
window.ITSupportTheme.setTheme(theme);
